cmake_minimum_required(VERSION 3.16.3)
project(fod)

set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(FONTDUMP_SOURCES fontdump.cpp)
set(GANIDUMP_SOURCES ganidump.cpp)
set(TILEDUMP_SOURCES tiledump.cpp)
set(UNPACK_SOURCES unpack.c)
set(SFX2WAV_SOURCES sfx2wav.c)

add_executable(fontdump ${FONTDUMP_SOURCES})
if(MSVC)
  target_compile_options(fontdump PRIVATE /W4 /WX)
else()
  target_compile_options(fontdump PRIVATE -Wall -Werror)
endif()

add_executable(unpack ${UNPACK_SOURCES})
if(MSVC)
  target_compile_options(unpack PRIVATE /W4 /WX)
else()
  target_compile_options(unpack PRIVATE -Wall -Werror)
endif()

add_executable(ganidump ${GANIDUMP_SOURCES})
target_include_directories(ganidump PRIVATE ${CMAKE_SOURCE_DIR}/src/lib)
target_link_libraries(ganidump PRIVATE common)
if(MSVC)
  target_compile_options(ganidump PRIVATE /W4 /WX)
else()
  target_compile_options(ganidump PRIVATE -Wall -Werror)
endif()

add_executable(tiledump ${TILEDUMP_SOURCES})
target_include_directories(tiledump PRIVATE ${CMAKE_SOURCE_DIR}/src/lib)
target_link_libraries(tiledump PRIVATE common)
if(MSVC)
  target_compile_options(tiledump PRIVATE /W4 /WX)
else()
  target_compile_options(tiledump PRIVATE -Wall -Werror)
endif()

list(APPEND SFX2WAV_LINK_LIBRARIES common)
if (NEED_LINKING_AGAINST_LIBM)
  list(APPEND SFX2WAV_LINK_LIBRARIES m)
endif()

add_executable(sfx2wav ${SFX2WAV_SOURCES})
target_include_directories(sfx2wav PRIVATE ${CMAKE_SOURCE_DIR}/src/lib)
target_link_libraries(sfx2wav PRIVATE ${SFX2WAV_LINK_LIBRARIES})
if(MSVC)
  target_compile_options(sfx2wav PRIVATE /W4 /WX)
else()
  target_compile_options(sfx2wav PRIVATE -Wall -Werror)
endif()
